```sql
-- Крупнейшие таблицы: данные/TOAST отдельно от индексов
SELECT schemaname, relname, n_live_tup AS estimated_rows,
       pg_size_pretty(pg_table_size(relid)) AS table_incl_toast,
       pg_size_pretty(pg_indexes_size(relid)) AS indexes,
       pg_size_pretty(pg_total_relation_size(relid)) AS total

ORDER BY pg_total_relation_size(relid) DESC

SELECT pg_size_pretty(pg_relation_size('public.embeddings'::regclass)) AS heap,
       pg_size_pretty(pg_table_size('public.embeddings'::regclass)) AS table_incl_toast,
       pg_size_pretty(pg_indexes_size('public.embeddings'::regclass)) AS indexes,
       pg_size_pretty(pg_total_relation_size('public.embeddings'::regclass)) AS total;

-- Тип интересующего столбца
SELECT column_name, data_type, udt_name
FROM information_schema.columns
WHERE table_schema = 'public' AND table_name = 'embeddings';

-- Приблизительная оценка логического размера значения на выборке;
-- это НЕ точная стоимость хранения столбца на диске.
SELECT COUNT(*) AS sampled_rows,
       AVG(pg_column_size(embedding))::numeric(12,2) AS avg_bytes,
       MAX(pg_column_size(embedding)) AS max_bytes
FROM public.embeddings TABLESAMPLE SYSTEM (0.1)
WHERE embedding IS NOT NULL;
```
